def verify(tag,c):
    rows=c['rows'];pos=[r for r in rows if r[1]>0]
    order=all((a[1]>b[1]) or (a[1]==b[1] and a[0]<b[0]) for a,b in zip(rows,rows[1:]))
    exp=[(PAL[i] if i<5 else ETC) for i in range(len(pos))]
    dots=[r[2] for r in pos]==exp and all(r[3]==rgb(r[2]) for r in rows) and all(r[2]==ZERO for r in rows if r[1]==0)
    seg=c['seg'];top=[s for s in seg if s[0]!='__etc'];etc=[s for s in seg if s[0]=='__etc']
    segok=[s[0] for s in top]==[r[0] for r in pos[:5]] and [s[1] for s in top]==PAL[:len(top)] and all(s[1]==s[2] for s in seg) and all(s[1]==ETC for s in etc) and (bool(etc)==(len(pos)>5))
    check(f'{tag}: 금액 내림차순(동률 ID 오름차순)',order,[(r[0][:10],r[1]) for r in rows])
    check(f'{tag}: 순위색 1~5 + 6위 이하·기타 #6B7280, 도넛=목록 점 일치',dots and segok,{'seg':[s[:2] for s in seg],'dots':[(r[0][:10],r[2]) for r in rows]})
