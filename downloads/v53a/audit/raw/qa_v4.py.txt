w2=lambda s:float(s.replace(',','').replace('원','').strip())
def verify(name,c,exp_total,exp_rows):
    tot=w2(c['total']);rs={r['key']:float(r['cost']) for r in c['rows']};psum=sum(float(r['pct'].rstrip('%')) for r in c['rows'])
    ok=abs(tot-exp_total)<0.01 and all(abs(rs.get(k,-1)-v)<0.01 for k,v in exp_rows.items()) and abs(sum(rs.values())-tot)<0.01 and (tot==0 or abs(psum-100)<=0.3)
    # pct numerator/denominator recompute
    pc=all(abs(float(r['pct'].rstrip('%'))-round(float(r['cost'])/tot*100,1))<0.051 for r in c['rows']) if tot else True
    check(name,ok and pc,{'total':c['total'],'rows':[(r['key'][:8],r['val'],r['pct']) for r in c['rows']],'pct_sum':round(psum,1),'expected_total':exp_total})
